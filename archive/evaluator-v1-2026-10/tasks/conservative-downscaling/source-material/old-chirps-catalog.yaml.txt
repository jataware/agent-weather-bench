obs/chirps-v3-monthly:
  adapter: http
  source_url: "https://data.chc.ucsb.edu/products/CHIRPS/v3.0/monthly/global/cogs/"
  file_pattern: "chirps-v3.0.{year}.{month:02d}.cog"
  format: cog
  request_interval: 3.0   # COG range reads; floor keeps bulk pulls under CrowdSec ~2 req/s
  variables:
    precip:
      native_name: precip
      units: "mm/month"
      target_units: "mm/day"
      fill_value: -9999
  grid:
    lat_res: 0.05
    lon_res: 0.05
    hindcast_range: [1981, 2026]   # final stream is ~1.5-month-lagged; fetch the current year with allow_partial=True
    temporal: monthly
  notes: "Native UCSB CHIRPS v3.0 monthly (0.05° COG). Final (gauge-corrected) stream; a new month posts by ~the 15th of the following month. For the near-real-time tail use obs/chirps-v3-monthly-prelim. Was obs/chirps-direct."
